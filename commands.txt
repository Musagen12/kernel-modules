Port fowarding
---------------
qemu-system-x86_64 \
  -kernel bzImage \
  -drive file=rootfs.img,format=raw \
  -append "console=ttyS0 nokaslr net.ifnames=0 root=/dev/sda rw" \
  -nographic \
  -m 2G \
  -netdev user,id=net0,hostfwd=tcp::2222-:22 \
  -device e1000,netdev=net0 \
  -serial mon:stdio


Internet setup
--------------
ip link set enp0s3 up
ip addr add 10.0.2.15/24 dev enp0s3
ip route add default via 10.0.2.2


Copying files into a qemu vm
-----------------------------
scp -P 2222 {filename}.ko root@localhost:/root/kernel-modules


SSH connection into the vm
---------------------------
ssh -p 2222 root@localhost
